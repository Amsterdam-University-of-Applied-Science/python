def get_company_info(ticker):
    try:
        # Create a Ticker object
        ticker_obj = yf.Ticker(ticker)

        # Get the company info
        info = ticker_obj.info

        # Extract company name and address
        company_name = info.get('longName', 'Name not available')
        company_address = info.get('address1', 'Address not available')
        company_city = info.get('city', 'City not available')
        company_state = info.get('state', 'State not available')
        company_zip = info.get('zip', 'Zip not available')
        company_country = info.get('country', 'Country not available')

        return {
            'name': company_name,
            'address': company_address,
            'city': company_city,
            'state': company_state,
            'zip': company_zip,
            'country': company_country
        }
    except Exception as e:
        return f"An error occurred: {str(e)}"

ticker = "TSLA"
company_info = get_company_info(ticker)  # Store the result in a variable

# Print company info in address format
print(f"{company_info['name']}\n{company_info['address']}\n{company_info['city']}\n{company_info['state']}, {company_info['zip']}\n{company_info['country']}")
